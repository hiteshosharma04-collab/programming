//  task 3
#include <iostream>
using namespace std;

class Task {
public:
    string title;
    bool isDone = false;

    Task(string t) { title = t; }

    void markDone() { isDone = true; }

    void display() {
        cout << title << " " << (isDone ? "DONE" : "PENDING");
    }
};

int main() {
    Task t("Study");
    t.markDone();
    t.display();
}


// task 4
#include <iostream>
#include <vector>
using namespace std;

class TaskList {
    vector<string> tasks;
public:
    void addTask(string t) { tasks.push_back(t); }

    void showTasks() {
        for(auto t : tasks) cout << t << endl;
    }
};

int main() {
    TaskList l;
    l.addTask("Study");
    l.addTask("Coding");
    l.showTasks();
}
